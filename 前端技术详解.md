### 核心功能模块

#### 1. 导航栏功能

```javascript
function initNavigation() {
  const navbar = document.getElementById('navbar');
  const hamburger = document.querySelector('.hamburger');
  const navMenu = document.querySelector('.nav-menu');
  
  // 滚动时改变导航栏样式
  window.addEventListener('scroll', function() {
    if (window.scrollY > 50) {
      navbar.classList.add('scrolled');
    } else {
      navbar.classList.remove('scrolled');
    }
  });
  
  // 移动端菜单切换
  hamburger.addEventListener('click', function() {
    navMenu.classList.toggle('active');
  });
  
  // 平滑滚动
  navLinks.forEach(link => {
    link.addEventListener('click', function(e) {
      e.preventDefault();
      const targetId = this.getAttribute('href');
      const targetSection = document.querySelector(targetId);
      targetSection.scrollIntoView({ behavior: 'smooth' });
    });
  });
}
```

**功能说明：**
- 滚动时导航栏样式变化
- 移动端汉堡菜单
- 点击导航链接平滑滚动

#### 2. 滚动动画

```javascript
function initScrollEffects() {
  const sections = document.querySelectorAll('.section');
  
  // Intersection Observer API
  const observer = new IntersectionObserver(function(entries) {
    entries.forEach(entry => {
      if (entry.isIntersecting) {
        entry.target.style.opacity = '1';
        entry.target.style.transform = 'translateY(0)';
      }
    });
  }, {
    threshold: 0.1,  // 元素10%可见时触发
    rootMargin: '0px 0px -100px 0px'
  });
  
  sections.forEach(section => {
    section.style.opacity = '0';
    section.style.transform = 'translateY(30px)';
    observer.observe(section);
  });
}
```

**技术点：**
- **Intersection Observer API** - 监听元素进入视口
- 性能优化：不需要频繁监听scroll事件

#### 3. 留言板功能

```javascript
function initMessageBoard() {
  const messageForm = document.getElementById('githubMessageForm');
  
  // 提交留言
  messageForm.addEventListener('submit', function(e) {
    e.preventDefault();  // 阻止表单默认提交
    
    const message = {
      name: this.name.value,
      email: this.email.value,
      content: this.content.value
    };
    
    // 发送到后端
    fetch(`${BACKEND_URL}/api/message`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(message)
    })
    .then(res => res.json())
    .then(response => {
      if (response.success) {
        showNotification('留言成功！', 'success');
        this.reset();
        loadMessages();  // 刷新留言列表
      }
    })
    .catch(err => {
      showNotification('留言失败：' + err.message, 'error');
    });
  });
  
  // 加载留言列表
  loadMessages();
}
```

**关键点：**
- `e.preventDefault()` - 阻止表单默认提交行为
- `fetch()` - 发送HTTP请求
- `JSON.stringify()` - 将对象转为JSON字符串
- `Promise.then()` - 处理异步响应

#### 4. 加载留言列表

```javascript
function loadMessages() {
  fetch(`${BACKEND_URL}/api/messages?t=${Date.now()}`, {
    headers: {
      'Cache-Control': 'no-cache'
    }
  })
  .then(res => res.json())
  .then(response => {
    const messages = response.data || [];
    
    // 清空现有内容
    messageList.innerHTML = '';
    
    // 渲染每条留言
    messages.forEach(msg => {
      const messageItem = document.createElement('div');
      messageItem.className = 'message-item';
      messageItem.innerHTML = `
        <p><strong>${escapeHtml(msg.name)}</strong></p>
        <p>${escapeHtml(msg.content)}</p>
        <p class="message-time">${formatTime(msg.createTime)}</p>
      `;
      messageList.appendChild(messageItem);
    });
  });
}
```

**技术点：**
- 动态创建DOM元素
- 防止XSS攻击：`escapeHtml()` 转义HTML
- 时间格式化

#### 5. 返回顶部按钮

## 前后端通信

### API调用流程

```
前端 (JavaScript)
   ↓
Fetch API
   ↓
HTTP请求 (JSON)
   ↓
后端 (Spring Boot)
   ↓
处理请求
   ↓
返回响应 (JSON)
   ↓
前端处理响应
   ↓
更新页面
```

### Fetch API 详解

#### 1. GET 请求（获取数据）

```javascript
fetch(`${BACKEND_URL}/api/messages`)
  .then(res => res.json())      // 将响应转为JSON
  .then(data => {
    // 处理数据
    console.log(data);
  })
  .catch(err => {
    // 处理错误
    console.error(err);
  });
```

#### 2. POST 请求（提交数据）

```javascript
fetch(`${BACKEND_URL}/api/message`, {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json'  // 告诉后端发送的是JSON
  },
  body: JSON.stringify({  // 将对象转为JSON字符串
    name: "张三",
    email: "zhangsan@example.com",
    content: "留言内容"
  })
})
.then(res => res.json())
.then(response => {
  if (response.success) {
    // 成功处理
  }
})
.catch(err => {
  // 错误处理
});
```

### 错误处理

```javascript
fetch(url)
  .then(res => {
    if (!res.ok) {
      throw new Error(`HTTP ${res.status}: ${res.statusText}`);
    }
    return res.json();
  })
  .then(data => {
    // 处理成功响应
  })
  .catch(err => {
    // 统一错误处理
    showNotification('操作失败：' + err.message, 'error');
  });
```

### 防止XSS攻击

```javascript
function escapeHtml(text) {
  const div = document.createElement('div');
  div.textContent = text;  // textContent会自动转义HTML
  return div.innerHTML;
}

// 使用
messageItem.innerHTML = `
  <p>${escapeHtml(userInput)}</p>  // 安全：HTML会被转义
`;
```

**为什么需要转义？**
- 防止恶意脚本注入
- 保护用户数据安全

```javascript
// 防抖：延迟执行，频繁触发时只执行最后一次
function debounce(func, wait) {
  let timeout;
  return function(...args) {
    clearTimeout(timeout);
    timeout = setTimeout(() => func.apply(this, args), wait);
  };
}

// 使用
```

## 性能优化技巧

### 1. 事件委托

```javascript
// ❌ 不好：给每个元素绑定事件
document.querySelectorAll('.item').forEach(item => {
  item.addEventListener('click', handleClick);
});

// ✅ 好：使用事件委托
document.querySelector('.container').addEventListener('click', function(e) {
  if (e.target.classList.contains('item')) {
    handleClick(e);
  }
});
```

### 2. 防抖滚动事件

```javascript
let ticking = false;

window.addEventListener('scroll', function() {
  if (!ticking) {
    window.requestAnimationFrame(function() {
      // 滚动处理逻辑
      ticking = false;
    });
    ticking = true;
  }
});
```

### 3. 图片懒加载

```javascript
const images = document.querySelectorAll('img[data-src]');

const imageObserver = new IntersectionObserver((entries) => {
  entries.forEach(entry => {

      const img = entry.target;

      imageObserver.unobserve(img);
    }
```

## 常见问题解决

### 1. CORS 跨域问题

**问题：** 前端无法访问后端API

**解决：** 后端已配置 `@CrossOrigin(origins = "*")`

### 2. 缓存问题

**问题：** 修改代码后页面没有更新

**解决：**
```javascript
// 添加时间戳防止缓存
fetch(`${BACKEND_URL}/api/messages?t=${Date.now()}`)
```

### 3. 异步问题

**问题：** 数据还没加载完就使用了

**解决：**
```javascript
// 使用 async/await
async function loadData() {
  const response = await fetch(url);
  const data = await response.json();
  // 使用数据
}
```
